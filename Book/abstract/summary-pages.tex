    \ifdim\dimexpr\ht\ETSITSummaryBox+\dp\ETSITSummaryBox\relax>

        {The Spanish and English summaries do not fit on one page at 9 pt}
        {Shorten abstract/resumen.tex or abstract/abstract.tex.}
    \fi
    \noindent\adjustbox{max totalheight=\textheight}{\usebox{\ETSITSummaryBox}}\par
  \endgroup
