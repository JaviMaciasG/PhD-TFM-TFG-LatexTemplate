% Render an empty line when the signer's DNI/NIF has not yet been configured.
\newcommand{\ConfidentialityDNI}[1]{%
  \ifthenelse{\equal{#1}{}}{\rule{3cm}{0.4pt}}{#1}}

% One complete copy of Annex VI for a professor who tutors or evaluates the TFG.
\newcommand{\ConfidentialityCommitment}[4]{%
  \begingroup
  \def\myConfidentialityProfessorFullName{#1}%
  \def\myConfidentialityProfessorDNI{#2}%
  \ifthenelse{\equal{#3}{male}}{%
    \def\wordDonOrDonaConfidentialityProfessor{D.}%
    \def\wordProfesorOrProfesoraConfidentiality{profesor}%
    \def\wordProfesorOrProfesoraConfidentialityTitle{Profesor}%
    \def\wordProfesorElOrLaConfidentiality{El}%
  }{%
    \def\wordDonOrDonaConfidentialityProfessor{Dª.}%
    \def\wordProfesorOrProfesoraConfidentiality{profesora}%
    \def\wordProfesorOrProfesoraConfidentialityTitle{Profesora}%
    \def\wordProfesorElOrLaConfidentiality{La}%
  }%

  \thispagestyle{empty}%

  \begin{center}
    \LARGE\bfseries COMPROMISO DE CONFIDENCIALIDAD EN TRABAJOS DE FIN DE GRADO (TFG)
  \end{center}

  \vspace{1.2cm}

  \wordDonOrDonaConfidentialityProfessor{} \myConfidentialityProfessorFullName{},
  con NIF \ConfidentialityDNI{\myConfidentialityProfessorDNI},
  \wordProfesorOrProfesoraConfidentiality{} de la Universidad de Alcalá, que #4
  el trabajo de fin de grado ``\myBookTitle'', presentado por
  \wordDonOrDonaAutor{} \myAuthorFullName{}, con NIF \myAuthorDNI{},

  \vspace{0.4cm}

  \noindent\textbf{SE COMPROMETE:}

  \begin{enumerate}[label=\Roman*.,leftmargin=1.1cm,itemsep=0.25cm,topsep=0.25cm]
    \item A no difundir ni por escrito ni oralmente el contenido del trabajo sin el
          consentimiento previo y por escrito del estudiante.

    \item A aplicar cuantas precauciones sean razonables para evitar la revelación a
          terceras personas.

    \item A no hacer servir el trabajo para ninguna otra finalidad que no sea la de
          su evaluación académica.

    \item A custodiar el trabajo durante el plazo exigido por la legislación vigente
          y, una vez transcurrido este plazo, proceder a su destrucción de acuerdo con
          el procedimiento establecido por la Universidad a estos efectos.

    \item A comunicar al estudiante cualquier hecho que pueda afectar a los
          compromisos de este documento.
  \end{enumerate}

  \vspace{0.2cm}

  \noindent\textbf{Conformes:}

  \vspace{0.5cm}

  \noindent\begin{tabularx}{\textwidth}{X X}
    \centering\textbf{\wordProfesorElOrLaConfidentiality{} \wordProfesorOrProfesoraConfidentialityTitle} &
    \centering\arraybackslash\textbf{\expandafter\makefirstuc\expandafter{\wordAutorElOrLa} \wordAlumnoOrAlumnaUpcaseFirt} \\
    \\[2.2cm]
    \centering Fdo.: \myConfidentialityProfessorFullName &
    \centering\arraybackslash Fdo.: \myAuthorFullName
  \end{tabularx}

  \vspace{0.6cm}

  \begin{center}
    Alcalá de Henares, a \myPaperworkDate{}.
  \end{center}
  \endgroup}
