  { Unsupported~degree~identifier~'#1'. }

\cs_new_protected:Npn \uah_degree_declare:nn #1#2
  {
    \seq_if_in:NnTF \g_uah_degree_registry_ids_seq {#1}
      { \msg_error:nnn { uah-degree-registry } { duplicate-degree } {#1} }
      {
        \uah_degree_clear_fields:
        \keys_set:nn { uah/degree } {#2}
        \uah_degree_require:nnN {#1} {status} \l_uah_degree_status_tl
        \uah_degree_require:nnN {#1} {institution} \l_uah_degree_institution_tl
        \uah_degree_require:nnN {#1} {degree-name} \l_uah_degree_name_tl
        \uah_degree_require:nnN {#1} {work-type} \l_uah_degree_work_type_tl
        \uah_degree_require:nnN {#1} {work-type-full-spanish} \l_uah_degree_work_type_full_spanish_tl
        \uah_degree_require:nnN {#1} {work-type-full-english} \l_uah_degree_work_type_full_english_tl
        \uah_degree_require:nnN {#1} {layout-profile} \l_uah_degree_layout_profile_tl
        \seq_gput_right:Nn \g_uah_degree_registry_ids_seq {#1}
        \uah_degree_store:nnV {#1} {status} \l_uah_degree_status_tl
        \uah_degree_store:nnV {#1} {institution} \l_uah_degree_institution_tl
        \uah_degree_store:nnV {#1} {degree-name} \l_uah_degree_name_tl
        \uah_degree_store:nnV {#1} {degree-name-english} \l_uah_degree_name_english_tl
        \uah_degree_store:nnV {#1} {degree-name-wrapped} \l_uah_degree_name_wrapped_tl
        \uah_degree_store:nnV {#1} {degree-name-wrapped-uppercase} \l_uah_degree_name_wrapped_uppercase_tl
        \uah_degree_store:nnV {#1} {work-type} \l_uah_degree_work_type_tl
        \uah_degree_store:nnV {#1} {work-type-full-spanish} \l_uah_degree_work_type_full_spanish_tl
        \uah_degree_store:nnV {#1} {work-type-full-english} \l_uah_degree_work_type_full_english_tl
        \uah_degree_store:nnV {#1} {layout-profile} \l_uah_degree_layout_profile_tl
      }
  }

  }

\cs_new_protected:Npn \uah_degree_select:n #1
  {
    \prop_if_in:NnTF \g_uah_degree_registry_prop {#1/work-type}
      {
        \uah_degree_validate_profile:n {#1}
        \tl_gset:Nn \g_uah_selected_degree_tl {#1}
        \uah_define_from_selected_property:Nn \myDegreefull {degree-name}
        \uah_define_from_selected_property:Nn \myWorkType {work-type}
        \uah_define_from_selected_property:Nn \myLayoutProfile {layout-profile}
        \tl_set:Nx \l_tmpa_tl {\myLanguage}
        \str_if_eq:VnTF \l_tmpa_tl {spanish}
          { \uah_define_from_selected_property:Nn \myWorkTypeFull {work-type-full-spanish} }
          { \uah_define_from_selected_property:Nn \myWorkTypeFull {work-type-full-english} }
        \prop_get:NnN \g_uah_degree_registry_prop {#1/degree-name-wrapped} \l_tmpa_tl
        \tl_if_blank:VTF \l_tmpa_tl
          { \cs_set:Npn \myDegreefullwrapped {\myDegreefull} }
          { \cs_set:Npx \myDegreefullwrapped { \exp_not:V \l_tmpa_tl } }
        \prop_get:NnN \g_uah_degree_registry_prop {#1/degree-name-wrapped-uppercase} \l_tmpa_tl
        \tl_if_blank:VTF \l_tmpa_tl
          { \cs_set:Npn \myDegreefullwrappedUpcase {\MakeUppercase{\myDegreefull}} }
          { \cs_set:Npx \myDegreefullwrappedUpcase { \exp_not:V \l_tmpa_tl } }
      }
      { \msg_error:nnn { uah-degree-registry } { unknown-degree } {#1} }
  }
